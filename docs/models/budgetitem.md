# BudgetItem

Represents a budget for tracking and notifying teams on their spending.

## Example Usage

```typescript
import { BudgetItem } from "@vercel/sdk/models/twohundredandeightyone.js";

let value: BudgetItem = {
  type: "fixed",
  fixedBudget: 2024.42,
  previousSpend: [
    5920.82,
    3690.02,
  ],
  notifiedAt: [
    7656.02,
  ],
  createdAt: 1086.99,
  isActive: true,
  teamId: "<id>",
  id: "<id>",
};
```
