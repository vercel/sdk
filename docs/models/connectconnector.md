# ConnectConnector

A connector that defines how Vercel accesses an external service.

## Example Usage

```typescript
import { ConnectConnector } from "@vercel/sdk/models/connectconnector.js";

let value: ConnectConnector = {
  createdAt: 636.06,
  displayName: "Natalie_Spencer",
  id: "<id>",
  name: "<value>",
  service: "<value>",
  supportedSubjectTypes: [
    "<value 1>",
  ],
  supportsIcon: "false",
  supportsInstallation: true,
  supportsRevocation: false,
  supportsTriggers: true,
  type: "linq",
  typeName: "<value>",
  uid: "<id>",
  updatedAt: 2319.25,
};
```
