# PresetConfiguration

## Example Usage

```typescript
import { PresetConfiguration } from "@vercel/sdk/models/connectcreateconnectorrequest.js";

let value: PresetConfiguration = {
  data: {},
  service: "<value>",
  connectionMethod: "<value>",
};
```
